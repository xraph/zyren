cmake_minimum_required(VERSION 3.18)
project(zyren_ml LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED ONNXRUNTIME_ROOT)
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the checksum-verified ONNX Runtime SDK.")
endif()
add_library(zyren_ml SHARED src/zyren_ml.cc)
target_include_directories(zyren_ml PRIVATE include "${ONNXRUNTIME_ROOT}/include")
set_target_properties(zyren_ml PROPERTIES CXX_VISIBILITY_PRESET hidden)
